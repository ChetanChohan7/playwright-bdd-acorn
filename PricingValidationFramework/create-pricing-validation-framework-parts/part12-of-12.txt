if not defined dotnetexe if exist "%USERPROFILE%\.dotnet\dotnet.exe" set "dotnetexe=%USERPROFILE%\.dotnet\dotnet.exe"

if not defined dotnetexe goto :nosdk
"%dotnetexe%" --list-sdks 2>nul | findstr /b "10." >nul
if errorlevel 1 goto :nosdk

echo Building with "%dotnetexe%"...
echo NuGet packages are downloaded to your user profile on the first build,
echo so this needs access to nuget.org or your company's package feed.
echo.
pushd "%targetdir%"
"%dotnetexe%" build PricingValidationFramework.slnx
set "buildresult=%errorlevel%"
popd
echo.
if not "%buildresult%"=="0" (
    echo The build failed - see the errors above. The source files are all in place.
    echo Common causes on locked-down machines: no access to nuget.org - ask for
    echo your company's NuGet feed - or the 260-character path limit.
    echo.
)
goto :nextsteps
